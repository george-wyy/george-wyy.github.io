% ---------------- Interests ----------------
\section{Research Interests}
Gaze-based interaction in AR/VR; gaze distance and depth ambiguity in near-eye target selection;
intent understanding from gaze, distinguishing selection, exploration, and confirmation;
eye--hand coordination and its computational (resource-rational) modeling;
head-mounted eye tracking, gaze calibration, and implicit recalibration from natural behavior;
target selection and multimodal (gaze + hand / head / foot) input.

\section{Research Experience}
\entry{Head-mounted binocular eye-tracking platform}{2025 -- present}{Lead developer (hardware and algorithms)}
\begin{itemize}
  \item Built a low-cost head-mounted binocular eye tracker end to end: 3D-printed frame, near-infrared eye cameras, scene camera, and real-time pipeline.
  \item Developed NIR pupil detection, gaze calibration (including mirror-based camera extrinsic calibration), binocular vergence depth, and multi-camera data collection tools.
  \item Using the platform to study eye--hand coordination and model-driven implicit calibration (PhD thesis).
\end{itemize}
\entry{Gaze interaction techniques for AR/VR}{2021 -- present}{First-author and co-author studies with Prof.\ Yafeng Niu's group}
\begin{itemize}
  \item Designed and evaluated gaze, blink, smooth-pursuit, gaze--head, gaze--foot, and gaze--hand selection techniques through controlled user studies (HoloLens~2, Unity, Tobii, Pupil Labs).
  \item Unobtrusive gaze calibration from mouse dragging (DragCal), with per-user eye--hand lag analysis.
\end{itemize}
\entry{Visual--haptic interaction mechanisms in gaze-controlled systems}{2023 -- Jun 2026 (completed)}{Technical lead, Ministry of Education Joint Fund project}
\begin{itemize}
  \item Led the technical roadmap, experimental design, and test-system implementation; the project passed final acceptance testing in 2026.
